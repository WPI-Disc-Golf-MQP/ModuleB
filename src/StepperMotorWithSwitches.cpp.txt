// Define pin connections & motor's steps per revolution
// defines pins numbers

int UPPER_LIMIT_SWITCH_PIN = 3;
int LOWER_LIMIT_SWITCH_PIN = 2;

const int stepPin = 8;
const int dirPin = 6;
const int sleepPin = 9;
 

void loop() {


  // Check if limit switch 1 is pressed (change direction)
  if (digitalRead(UPPER_LIMIT_SWITCH_PIN) == LOW) {
    digitalWrite(dirPin, !digitalRead(dirPin)); // Toggle the direction
    delay(500); // Debounce delay(500)
    return;
  }

  // Check if lower limit switch is pressed (stop the motor)
  if (digitalRead(LOWER_LIMIT_SWITCH_PIN) == LOW) {
    digitalWrite(sleepPin, !digitalRead(sleepPin));
    // Stop the motor (do nothing)
    return;
  }
  // Continue pulsing the step pin to move the motor
  for (int i = 0; i < 100; i++) {

  digitalWrite(stepPin, HIGH);
  delayMicroseconds(1000); //(500)
  digitalWrite(stepPin, LOW);
  delayMicroseconds(1000);
  }
}

void setup() {
  // Sets the two pins as Outputs
  pinMode(stepPin, OUTPUT);
  pinMode(dirPin, OUTPUT);
  pinMode(sleepPin, OUTPUT);
  
  digitalWrite(stepPin, HIGH); // Initalize with Movement
  digitalWrite(dirPin, LOW); // Direction of Turn
  digitalWrite(sleepPin, HIGH); //A logic high allows normal operation of the A4988 by removing from sleep

//switches set as inputs
pinMode (UPPER_LIMIT_SWITCH_PIN, INPUT_PULLUP);
pinMode (LOWER_LIMIT_SWITCH_PIN, INPUT_PULLUP);

}
